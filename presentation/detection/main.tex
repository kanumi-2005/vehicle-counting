\begin{frame}

\framesubtitle{Giới thiệu}

\begin{itemize}[<+->]
  \item Mô hình phát hiện đối tượng thời gian thực thuộc họ YOLO
  \item Sử dụng kiến trúc Anchor-free
  \item Cải tiến Backbone và Neck để trích xuất đặc trưng hiệu quả hơn
  \item Cân bằng tốt giữa tốc độ và độ chính xác
  \item Hỗ trợ đa tác vụ
\end{itemize}

\begin{frame}
\frametitle{YOLOv8n}
\framesubtitle{Quá trình fine-tuning}

\begin{itemize}[<+->]
  \item GPU: NVIDIA Tesla T4 (Google Colab)
  \item Bộ dữ liệu: UA-DETRAC
  \item Tỉ lệ chia dữ liệu: 80\% train--20\% test
  \item Trọng số khởi tạo YOLOv8n pretrained trên COCO
  \item Đóng băng 10 lớp đầu tiên
  \item Epoch: 20
  \item Batch size: 64
\end{itemize}

\end{frame}
