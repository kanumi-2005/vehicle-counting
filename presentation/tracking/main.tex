\section{Theo dõi phương tiện}

\begin{frame}
\frametitle{BYTE}
\framesubtitle{Tổng quan}

\begin{itemize}[<+->]
  \item Thuật toán theo dõi đa đối tượng theo hướng tracking-by-detection
  \item Nhận đầu vào là các hộp phát hiện từ YOLOv8n ở từng khung hình
  \item Duy trì định danh phương tiện bằng Kalman Filter và Hungarian matching
  \item Khác SORT ở việc tận dụng cả các phát hiện có độ tin cậy thấp
  \item Phù hợp với bối cảnh giao thông có che khuất và phương tiện nhỏ
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{BYTE}
\framesubtitle{Cơ chế liên kết dữ liệu}

\begin{itemize}[<+->]
  \item Chia các phát hiện thành hai nhóm: độ tin cậy cao và độ tin cậy thấp
  \item Liên kết các phát hiện độ tin cậy cao với các vết theo dõi hiện có
  \item Dùng các phát hiện độ tin cậy thấp để khôi phục các vết chưa được gán
  \item Tạo vết theo dõi mới từ các phát hiện độ tin cậy cao còn lại
  \item Giảm mất dấu và hạn chế thay đổi ID khi phương tiện bị che khuất
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Nhận xét}
\framesubtitle{Ưu điểm và hạn chế}

\begin{itemize}[<+->]
  \item YOLOv8n + BYTE là hướng tiếp cận khả thi cho theo dõi phương tiện
  \item BYTE tận dụng phát hiện độ tin cậy thấp để giảm mất dấu trong video
  \item Hệ thống đơn giản, dễ thay thế detector hoặc tracker khi cần
  \item Hiệu năng vẫn phụ thuộc mạnh vào chất lượng hộp phát hiện từ YOLO
  \item Đối tượng nhỏ, che khuất hoặc định vị chưa chính xác làm giảm DetA và HOTA
\end{itemize}

\end{frame}
